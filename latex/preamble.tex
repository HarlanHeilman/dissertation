% WSU: at least 1 in margins, page numbers centered 0.5 in from the bottom edge.
% Side margins of 1.4 in give a 5.7 in measure, about 80 characters per line (WSU: at least 1 in margins, and a
% measure of 45 to 90 characters).
% Running heads sit in the top margin, their baseline 0.7 in from the top edge (WSU exempts running heads
% from the 1 in margin and asks only that nothing come within 0.5 in of the edge).
% Page numbers: the Graduate School review asks for at least 0.5 in of blank space between the bottom of the text
% and the top of the page number, with the number centered 0.5 in from the bottom edge. A footskip of 0.7 in
% (baseline to baseline) leaves more than 0.5 in of clear space above the digits, and a bottom margin of 1.2 in puts the
% number's baseline 0.5 in from the edge, outside the 0.5 in absolute margin.

% ---------------- headings and lists, following the WSU ETD example (2024) ----------------
\usepackage{titlesec}
% Letter-spaced small capitals for display headings and running heads (fontspec tracking, works in LuaLaTeX and XeLaTeX).
\newcommand{\spacedsc}[1]{{\addfontfeatures{LetterSpace=6}\scshape #1}}
\newcommand{\spacedlowsc}[1]{{\addfontfeatures{LetterSpace=6}\scshape\MakeLowercase{#1}}}
% Numbered chapters: a spaced small-capital "Chapter" over a large numeral, the title in spaced small capitals,
% and a short rule. Every chapter opens the same way, as WSU requires.
\titleformat{\chapter}[display]
  {\normalfont\centering}
  {\spacedlowsc{\chaptertitlename}\\[4pt]{\fontsize{44}{44}\selectfont\thechapter}}
  {14pt}
  {\Large\spacedsc}
  [\vspace{10pt}\rule{0.25\textwidth}{0.4pt}]
\titlespacing*{\chapter}{0pt}{0pt}{28pt}
% Unnumbered chapter-level headings (Acknowledgments, Dedication, References, lists): centered capitals at body size.
\titleformat{name=\chapter,numberless}[block]{\normalfont\normalsize\centering}{}{0pt}{\MakeUppercase}
\titlespacing*{name=\chapter,numberless}{0pt}{0pt}{12pt}
% Section levels: spaced small capitals, then bold, then italic.
\titleformat{\section}{\normalfont\large}{\thesection}{0.9em}{\spacedsc}
\titleformat{\subsection}{\normalfont\normalsize\bfseries}{\thesubsection}{0.8em}{}
\titleformat{\subsubsection}{\normalfont\normalsize\itshape}{\thesubsubsection}{0.8em}{}
\titlespacing*{\section}{0pt}{2.4ex plus .5ex minus .2ex}{1.2ex plus .2ex}
\titlespacing*{\subsection}{0pt}{1.6ex plus .4ex minus .2ex}{0.6ex plus .2ex}
\titlespacing*{\subsubsection}{0pt}{1.4ex plus .4ex minus .2ex}{0.4ex}
